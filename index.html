Introduction to JSDoc in ServiceNow

JSDoc is a powerful documentation system for JavaScript that allows you to annotate your code with structured comments. In ServiceNow, proper documentation is crucial for maintaining enterprise-scale applications, enabling team collaboration, and ensuring code maintainability.

💡 Why Use JSDoc in ServiceNow?
  • Improve code readability and maintainability
  • Enable better IDE autocomplete and IntelliSense
  • Generate automatic documentation
  • Facilitate team collaboration
  • Document complex ServiceNow APIs and workflows

JSDoc comments begin with /** and end with */. Each line typically starts with an asterisk * for consistency.

/**
 * This is a JSDoc comment
 * It can span multiple lines
 */

JSDoc Basics

Basic Function Documentation

Every function should be documented with a description, parameters, and return value.

/**
 * Calculates the total price including tax
 * @param {number} basePrice - The base price before tax
 * @param {number} taxRate - The tax rate as a decimal (e.g., 0.08 for 8%)
 * @returns {number} The total price including tax
 * @example
 * const total = calculateTotal(100, 0.08);
 * // Returns: 108
 */
function calculateTotal(basePrice, taxRate) {
    return basePrice * (1 + taxRate);
}

Class Documentation

Classes should document their purpose, constructor parameters, and public methods.

/**
 * Represents a ServiceNow Incident Manager
 * @class
 * @classdesc Manages incident creation, updates, and resolution workflows
 */
var IncidentManager = Class.create();
IncidentManager.prototype = {
    /**
     * Initializes the Incident Manager
     * @constructor
     * @param {GlideRecord} incidentRecord - The incident record to manage
     */
    initialize: function(incidentRecord) {
        this.incident = incidentRecord;
        this.logger = new GSLog('IncidentManager', 'global');
    },

    /**
     * Assigns the incident to a user or group
     * @param {string} assignee - sys_id of the user or group
     * @param {string} type - Either 'user' or 'group'
     * @returns {boolean} True if assignment was successful
     * @throws {Error} If assignee type is invalid
     */
    assignIncident: function(assignee, type) {
        if (type !== 'user' && type !== 'group') {
            throw new Error('Invalid assignee type: ' + type);
        }

        if (type === 'user') {
            this.incident.assigned_to = assignee;
        } else {
            this.incident.assignment_group = assignee;
        }

        return this.incident.update();
    },

    type: 'IncidentManager'
};

ServiceNow-Specific Documentation

Business Rules

Business Rules should document their trigger conditions and expected behavior.

/**
 * Business Rule: Auto-assign Incidents
 * 
 * @description Automatically assigns incidents to the appropriate group based on category
 * @trigger Before Insert
 * @table incident
 * @conditions Category is not empty
 * @author ServiceNow Admin
 * @since 2024.01
 */
(function executeRule(current, previous /*null when async*/) {
    /**
     * Gets the assignment group based on incident category
     * @param {string} category - The incident category
     * @returns {string} sys_id of the assignment group
     */
    function getAssignmentGroup(category) {
        var gr = new GlideRecord('sys_user_group');
        gr.addQuery('name', 'CONTAINS', category);
        gr.query();
        
        if (gr.next()) {
            return gr.sys_id.toString();
        }
        
        return '';
    }

    // Only process if category is set
    if (!current.category.nil()) {
        var groupId = getAssignmentGroup(current.category.toString());
        
        if (groupId) {
            current.assignment_group = groupId;
            gs.info('Auto-assigned incident ' + current.number + ' to group: ' + groupId);
        }
    }
})(current, previous);

Script Include

Script Includes should thoroughly document their API and usage.

/**
 * ServiceNow Utility Library for Incident Management
 * 
 * @class IncidentUtils
 * @classdesc Provides utility functions for incident management operations
 * @description This Script Include contains helper methods for common incident operations
 *              including priority calculation, SLA management, and notification handling
 * 
 * @author ServiceNow Team
 * @since Rome Release
 * @version 2.1.0
 * 
 * @example
 * var utils = new IncidentUtils();
 * var priority = utils.calculatePriority(3, 2);
 */
var IncidentUtils = Class.create();
IncidentUtils.prototype = {
    /**
     * Initializes the IncidentUtils class
     * @constructor
     */
    initialize: function() {
        this.PRIORITY_MATRIX = {
            '1-1': 1, '1-2': 1, '1-3': 2,
            '2-1': 1, '2-2': 2, '2-3': 3,
            '3-1': 2, '3-2': 3, '3-3': 4
        };
    },

    /**
     * Calculates incident priority based on urgency and impact
     * 
     * @param {number} urgency - Urgency level (1-3, where 1 is highest)
     * @param {number} impact - Impact level (1-3, where 1 is highest)
     * @returns {number} Calculated priority (1-4)
     * 
     * @throws {TypeError} If urgency or impact are not numbers
     * @throws {RangeError} If urgency or impact are outside 1-3 range
     * 
     * @example
     * var priority = utils.calculatePriority(1, 2);
     * // Returns: 1 (Critical)
     * 
     * @example
     * var priority = utils.calculatePriority(3, 3);
     * // Returns: 4 (Low)
     */
    calculatePriority: function(urgency, impact) {
        // Validate inputs
        if (typeof urgency !== 'number' || typeof impact !== 'number') {
            throw new TypeError('Urgency and impact must be numbers');
        }

        if (urgency < 1 || urgency > 3 || impact < 1 || impact > 3) {
            throw new RangeError('Urgency and impact must be between 1 and 3');
        }

        var key = urgency + '-' + impact;
        return this.PRIORITY_MATRIX[key] || 4;
    },

    /**
     * Sends notification to incident stakeholders
     * 
     * @param {string} incidentSysId - sys_id of the incident record
     * @param {string} notificationType - Type of notification to send
     * @param {Object} [additionalData={}] - Additional data for notification template
     * @param {string} additionalData.customMessage - Custom message to include
     * @param {Array} additionalData.recipients - Additional recipient sys_ids
     * 
     * @returns {boolean} True if notification was sent successfully
     * 
     * @example
     * var success = utils.sendNotification(
     *     'abc123',
     *     'incident.assigned',
     *     {
     *         customMessage: 'Please review urgently',
     *         recipients: ['user1_sys_id', 'user2_sys_id']
     *     }
     * );
     */
    sendNotification: function(incidentSysId, notificationType, additionalData) {
        additionalData = additionalData || {};
        
        var gr = new GlideRecord('incident');
        if (!gr.get(incidentSysId)) {
            gs.error('Incident not found: ' + incidentSysId);
            return false;
        }

        try {
            gs.eventQueue(notificationType, gr, additionalData.customMessage, additionalData.recipients);
            return true;
        } catch (e) {
            gs.error('Failed to send notification: ' + e.message);
            return false;
        }
    },

    /**
     * Retrieves incident statistics for a given date range
     * 
     * @param {string} startDate - Start date in format 'YYYY-MM-DD'
     * @param {string} endDate - End date in format 'YYYY-MM-DD'
     * @param {Object} [filters={}] - Optional filters
     * @param {string} filters.category - Filter by category
     * @param {string} filters.state - Filter by state
     * @param {string} filters.assignmentGroup - Filter by assignment group sys_id
     * 
     * @returns {IncidentStats} Object containing incident statistics
     * 
     * @typedef {Object} IncidentStats
     * @property {number} total - Total number of incidents
     * @property {number} open - Number of open incidents
     * @property {number} resolved - Number of resolved incidents
     * @property {number} avgResolutionTime - Average resolution time in hours
     * @property {Object} byPriority - Breakdown by priority level
     * 
     * @example
     * var stats = utils.getIncidentStats('2024-01-01', '2024-01-31', {
     *     category: 'Network',
     *     state: '1'
     * });
     * gs.info('Total incidents: ' + stats.total);
     */
    getIncidentStats: function(startDate, endDate, filters) {
        filters = filters || {};
        
        var gr = new GlideRecord('incident');
        gr.addQuery('opened_at', '>=', startDate);
        gr.addQuery('opened_at', '<=', endDate);
        
        if (filters.category) {
            gr.addQuery('category', filters.category);
        }
        if (filters.state) {
            gr.addQuery('state', filters.state);
        }
        if (filters.assignmentGroup) {
            gr.addQuery('assignment_group', filters.assignmentGroup);
        }
        
        gr.query();
        
        var stats = {
            total: gr.getRowCount(),
            open: 0,
            resolved: 0,
            avgResolutionTime: 0,
            byPriority: { '1': 0, '2': 0, '3': 0, '4': 0 }
        };
        
        var totalResolutionTime = 0;
        var resolvedCount = 0;
        
        while (gr.next()) {
            // Count by state
            var state = gr.state.toString();
            if (state === '6' || state === '7') {
                stats.resolved++;
                
                // Calculate resolution time
                var opened = new GlideDateTime(gr.opened_at);
                var resolved = new GlideDateTime(gr.resolved_at);
                var diff = GlideDateTime.subtract(opened, resolved);
                totalResolutionTime += diff.getNumericValue();
                resolvedCount++;
            } else {
                stats.open++;
            }
            
            // Count by priority
            var priority = gr.priority.toString();
            if (stats.byPriority[priority] !== undefined) {
                stats.byPriority[priority]++;
            }
        }
        
        if (resolvedCount > 0) {
            stats.avgResolutionTime = Math.round(totalResolutionTime / resolvedCount / 3600000);
        }
        
        return stats;
    },

    /**
     * @private
     * @description Internal helper method - do not use directly
     */
    _validateRecord: function(record) {
        return record && record.isValidRecord();
    },

    type: 'IncidentUtils'
};

Client Script

Client Scripts should document their type and form interactions.

/**
 * Client Script: Validate Incident Priority
 * 
 * @description Validates incident priority against urgency and impact on form submission
 * @type onChange
 * @table incident
 * @field urgency, impact
 * 
 * @param {Object} control - The form control that triggered the event
 * @param {Object} oldValue - Previous field value
 * @param {Object} newValue - New field value
 * @param {boolean} isLoading - True if form is loading
 * @param {boolean} isTemplate - True if this is a template
 */
function onChange(control, oldValue, newValue, isLoading, isTemplate) {
    if (isLoading || isTemplate) {
        return;
    }

    /**
     * Calculates the correct priority based on urgency and impact
     * @param {string} urgency - Urgency value
     * @param {string} impact - Impact value
     * @returns {string} Calculated priority value
     */
    function calculatePriority(urgency, impact) {
        var priorityMatrix = {
            '1-1': '1', '1-2': '1', '1-3': '2',
            '2-1': '1', '2-2': '2', '2-3': '3',
            '3-1': '2', '3-2': '3', '3-3': '4'
        };
        
        var key = urgency + '-' + impact;
        return priorityMatrix[key] || '4';
    }

    var urgency = g_form.getValue('urgency');
    var impact = g_form.getValue('impact');
    
    if (urgency && impact) {
        var calculatedPriority = calculatePriority(urgency, impact);
        g_form.setValue('priority', calculatedPriority);
    }
}

UI Action

/**
 * UI Action: Escalate Incident
 * 
 * @description Escalates an incident to the next priority level and notifies management
 * @condition current.state != '6' && current.state != '7'
 * @action Button
 * @table incident
 * 
 * @returns {void}
 */
(function() {
    /**
     * Escalates the current incident
     * @param {GlideRecord} incident - The incident to escalate
     * @returns {boolean} Success status
     */
    function escalateIncident(incident) {
        var currentPriority = parseInt(incident.priority.toString());
        
        if (currentPriority <= 1) {
            gs.addErrorMessage('Incident is already at highest priority');
            return false;
        }
        
        // Decrease priority number (increases urgency)
        incident.priority = (currentPriority - 1).toString();
        incident.escalation = 1; // Set escalation flag
        incident.update();
        
        // Send notification to management
        gs.eventQueue('incident.escalated', incident);
        
        gs.addInfoMessage('Incident escalated to Priority ' + incident.priority);
        return true;
    }
    
    escalateIncident(current);
    action.setRedirectURL(current);
})();

Generic Types and Templates

/**
 * Generic repository for ServiceNow table operations
 * 
 * @template T
 * @class Repository
 * @classdesc Provides CRUD operations for any ServiceNow table
 */
var Repository = Class.create();
Repository.prototype = {
    /**
     * @constructor
     * @param {string} tableName - Name of the ServiceNow table
     */
    initialize: function(tableName) {
        this.tableName = tableName;
    },

    /**
     * Finds a record by sys_id
     * 
     * @template T
     * @param {string} sysId - Record sys_id
     * @returns {T|null} The record if found, null otherwise
     */
    findById: function(sysId) {
        var gr = new GlideRecord(this.tableName);
        if (gr.get(sysId)) {
            return this._recordToObject(gr);
        }
        return null;
    },

    /**
     * Finds all records matching criteria
     * 
     * @template T
     * @param {Object} criteria - Search criteria
     * @param {number} [limit=100] - Maximum number of results
     * @returns {Array} Array of matching records
     */
    findAll: function(criteria, limit) {
        limit = limit || 100;
        var gr = new GlideRecord(this.tableName);
        
        for (var field in criteria) {
            gr.addQuery(field, criteria[field]);
        }
        
        gr.setLimit(limit);
        gr.query();
        
        var results = [];
        while (gr.next()) {
            results.push(this._recordToObject(gr));
        }
        
        return results;
    },

    /**
     * @private
     * @param {GlideRecord} gr - GlideRecord to convert
     * @returns {Object} Plain object representation
     */
    _recordToObject: function(gr) {
        var obj = {};
        var fields = gr.getFields();
        
        for (var i = 0; i < fields.size(); i++) {
            var field = fields.get(i);
            obj[field.getName()] = gr.getValue(field.getName());
        }
        
        return obj;
    },

    type: 'Repository'
};

Async Operations and Promises

/**
 * @callback AsyncCallback
 * @param {Error|null} error - Error object if operation failed, null otherwise
 * @param {*} result - Operation result
 */

/**
 * Asynchronously fetches incident data from external system
 * 
 * @param {string} incidentNumber - Incident number to fetch
 * @param {AsyncCallback} callback - Callback function
 * 
 * @returns {void}
 * 
 * @example
 * fetchExternalIncident('INC0010001', function(error, data) {
 *     if (error) {
 *         gs.error('Failed to fetch incident: ' + error.message);
 *         return;
 *     }
 *     
 *     gs.info('Fetched incident: ' + JSON.stringify(data));
 * });
 */
function fetchExternalIncident(incidentNumber, callback) {
    var request = new sn_ws.RESTMessageV2();
    request.setEndpoint('https://api.example.com/incidents/' + incidentNumber);
    request.setHttpMethod('GET');
    
    request.executeAsync(function(response) {
        if (response.getStatusCode() === 200) {
            try {
                var data = JSON.parse(response.getBody());
                callback(null, data);
            } catch (e) {
                callback(new Error('Failed to parse response: ' + e.message));
            }
        } else {
            callback(new Error('HTTP ' + response.getStatusCode()));
        }
    });
}
